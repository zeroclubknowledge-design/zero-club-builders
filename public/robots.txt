# Zero Club
#
# Everything public is crawlable. The favicon is called out explicitly because
# Google will not show a site icon in search results if it cannot fetch it, and
# a blocked icon is one of the few ways to end up with the generic globe even
# when the file exists.

User-agent: *
Allow: /
Allow: /favicon.ico

# Signed-in areas hold nothing useful to a crawler and should not be indexed.
Disallow: /app/
# Shared posts have a public, signed-out view and social preview metadata.
Allow: /app/post/
# Public shared pages must remain available to social preview crawlers.
Allow: /app/bootcamps/
Allow: /app/notes/
Allow: /app/profile/
Disallow: /app/profile/edit
# Link-preview images live under /api, and some chat apps check robots.txt
# before fetching a preview image.
Allow: /api/og/
Allow: /api/link-preview
Allow: /api/og-default
Allow: /api/club-image/
Allow: /api/game-image/
Allow: /api/gift-image/
Disallow: /api/
